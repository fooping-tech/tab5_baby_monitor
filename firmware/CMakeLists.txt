cmake_minimum_required(VERSION 3.16)
set(ESP_DL_PATH "$ENV{ESP_DL_PATH}" CACHE PATH "Pinned ESP-DL checkout")
if(NOT EXISTS "${ESP_DL_PATH}/models/yolo26/yolo26.hpp")
    message(FATAL_ERROR "Set ESP_DL_PATH to the ESP-DL checkout documented in README.md")
endif()
execute_process(COMMAND git -C "${ESP_DL_PATH}" rev-parse HEAD
                OUTPUT_VARIABLE ESP_DL_REV OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE GIT_RESULT)
if(NOT GIT_RESULT EQUAL 0 OR NOT ESP_DL_REV STREQUAL "5d9c36063dddbe98b5387828c831d6bbadb1370f")
    message(FATAL_ERROR "ESP-DL revision mismatch; use the pinned revision")
endif()
set(EXTRA_COMPONENT_DIRS "${ESP_DL_PATH}/esp-dl" "${ESP_DL_PATH}/models/yolo26")
set(TAB5_REFERENCE_PATH "$ENV{TAB5_REFERENCE_PATH}" CACHE PATH "Pinned tab5_rtsp_logger checkout")
execute_process(COMMAND git -C "${TAB5_REFERENCE_PATH}" rev-parse HEAD
                OUTPUT_VARIABLE TAB5_REV OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE TAB5_GIT_RESULT)
if(NOT TAB5_GIT_RESULT EQUAL 0 OR NOT TAB5_REV STREQUAL "e5e5fee79c2232bd5de4a994d8f90e12f4630952")
    message(FATAL_ERROR "Set TAB5_REFERENCE_PATH to tab5_rtsp_logger revision e5e5fee79c2232bd5de4a994d8f90e12f4630952")
endif()
foreach(component esp_video esp_cam_sensor esp_sccb_intf)
    list(APPEND EXTRA_COMPONENT_DIRS "${TAB5_REFERENCE_PATH}/platforms/tab5/components/${component}")
endforeach()
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(tab5_baby_edge_ai)
