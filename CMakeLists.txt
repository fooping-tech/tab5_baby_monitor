cmake_minimum_required(VERSION 3.16)
project(baby_edge_core LANGUAGES CXX)
enable_testing()
add_executable(presence_test tests/presence_test.cpp)
target_include_directories(presence_test PRIVATE core)
target_compile_features(presence_test PRIVATE cxx_std_17)
target_compile_options(presence_test PRIVATE -Wall -Wextra -Werror)
add_test(NAME presence COMMAND presence_test)
add_executable(rtsp_framing_test tests/rtsp_framing_test.cpp firmware/main/rtsp/rtsp_framing.cpp)
target_include_directories(rtsp_framing_test PRIVATE firmware/main/rtsp)
target_compile_features(rtsp_framing_test PRIVATE cxx_std_17)
target_compile_options(rtsp_framing_test PRIVATE -Wall -Wextra -Werror)
add_test(NAME rtsp_framing COMMAND rtsp_framing_test)
add_executable(frame_convert_test tests/frame_convert_test.cpp)
target_include_directories(frame_convert_test PRIVATE core)
target_compile_features(frame_convert_test PRIVATE cxx_std_17)
target_compile_options(frame_convert_test PRIVATE -Wall -Wextra -Werror)
add_test(NAME frame_convert COMMAND frame_convert_test)
