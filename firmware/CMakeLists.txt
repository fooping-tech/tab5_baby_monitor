cmake_minimum_required(VERSION 3.16)
set(TAB5_REFERENCE_PATH "$ENV{TAB5_REFERENCE_PATH}" CACHE PATH "Pinned tab5_rtsp_logger checkout")
execute_process(COMMAND git -C "${TAB5_REFERENCE_PATH}" rev-parse HEAD
                OUTPUT_VARIABLE TAB5_REV OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE TAB5_GIT_RESULT)
if(NOT TAB5_GIT_RESULT EQUAL 0 OR NOT TAB5_REV STREQUAL "e5e5fee79c2232bd5de4a994d8f90e12f4630952")
    message(FATAL_ERROR "Set TAB5_REFERENCE_PATH to tab5_rtsp_logger revision e5e5fee79c2232bd5de4a994d8f90e12f4630952")
endif()
foreach(component esp_video esp_cam_sensor esp_sccb_intf m5stack_tab5 esp_lcd_st7121 espressif__esp_lvgl_port)
    list(APPEND EXTRA_COMPONENT_DIRS "${TAB5_REFERENCE_PATH}/platforms/tab5/components/${component}")
endforeach()
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(tab5_camera_clock)
if(CONFIG_CODEC_I2C_BACKWARD_COMPATIBLE)
    message(FATAL_ERROR "Disable CODEC_I2C_BACKWARD_COMPATIBLE: Tab5 BSP requires the new I2C driver")
endif()
